import base64
import json
import sys
from pathlib import Path
from PIL import Image

folder = Path(__file__).resolve().parent
session = Path('C:/Users/97349/.codex/sessions/2026/10/21/rollout-2026-10-21T00-33-48-01a15a29-7116-7a22-91e0-c4330740cfcd.jsonl')
name, prefix = sys.argv[1:3]
matches = []
for line in session.open(encoding='utf-8'):
    if '"image_generation_call"' not in line:
        continue
    record = json.loads(line)
    item = record.get('payload', {})
    if item.get('type') == 'image_generation_call' and item.get('status') == 'completed' and item.get('revised_prompt', '').startswith(prefix):
        matches.append((record.get('timestamp'), item))
if not matches:
    raise SystemExit('Expected completed native result not found')
timestamp, item = matches[-1]
target = folder / (name + '-native.png')
if target.exists():
    raise SystemExit('Refusing to overwrite existing result')
target.write_bytes(base64.b64decode(item['result']))
with Image.open(target) as im:
    info = {'id': item['id'], 'timestamp': timestamp, 'size': im.size, 'mode': im.mode,
            'alphaRange': im.getchannel('A').getextrema() if 'A' in im.getbands() else None,
            'prompt': item['revised_prompt'], 'nativeFile': str(target)}
(folder / (name + '-record.json')).write_text(json.dumps(info, ensure_ascii=False, indent=2), encoding='utf-8')
print(json.dumps({k: v for k, v in info.items() if k != 'prompt'}, ensure_ascii=False))
